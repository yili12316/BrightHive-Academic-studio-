-- BrightHive Academic Studio production parent-portal schema.
-- Run once in Supabase Dashboard -> SQL Editor.
-- No sample users or sample school records are created by this script.

create extension if not exists pgcrypto;

create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text not null default '',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.students (
  id uuid primary key default gen_random_uuid(),
  student_number text unique,
  first_name text not null,
  last_name text not null,
  grade text not null,
  status text not null default 'active' check (status in ('active','inactive')),
  created_at timestamptz not null default now()
);

create table if not exists public.parent_students (
  parent_id uuid not null references auth.users(id) on delete cascade,
  student_id uuid not null references public.students(id) on delete cascade,
  relationship text not null default 'Parent',
  primary key (parent_id, student_id)
);

create table if not exists public.classes (
  id uuid primary key default gen_random_uuid(),
  code text unique,
  subject text not null,
  title text not null,
  teacher_name text not null,
  room text,
  term text,
  color text not null default 'green' check (color in ('green','blue','orange','purple')),
  created_at timestamptz not null default now()
);

create table if not exists public.enrollments (
  student_id uuid not null references public.students(id) on delete cascade,
  class_id uuid not null references public.classes(id) on delete cascade,
  progress smallint not null default 0 check (progress between 0 and 100),
  primary key (student_id, class_id)
);

create table if not exists public.schedule_items (
  id uuid primary key default gen_random_uuid(),
  class_id uuid not null references public.classes(id) on delete cascade,
  weekday smallint not null check (weekday between 1 and 7),
  start_time time not null,
  end_time time not null
);

create table if not exists public.assignments (
  id uuid primary key default gen_random_uuid(),
  class_id uuid not null references public.classes(id) on delete cascade,
  title text not null,
  due_at timestamptz,
  total_points numeric(8,2),
  published_at timestamptz not null default now()
);

create table if not exists public.assignment_progress (
  student_id uuid not null references public.students(id) on delete cascade,
  assignment_id uuid not null references public.assignments(id) on delete cascade,
  status text not null default 'assigned' check (status in ('assigned','in_progress','submitted','graded')),
  progress smallint not null default 0 check (progress between 0 and 100),
  score numeric(8,2),
  primary key (student_id, assignment_id)
);

create table if not exists public.exams (
  id uuid primary key default gen_random_uuid(),
  class_id uuid not null references public.classes(id) on delete cascade,
  title text not null,
  exam_date date not null,
  max_score numeric(8,2) not null default 100
);

create table if not exists public.exam_results (
  student_id uuid not null references public.students(id) on delete cascade,
  exam_id uuid not null references public.exams(id) on delete cascade,
  score numeric(8,2),
  teacher_comment text,
  published_at timestamptz not null default now(),
  primary key (student_id, exam_id)
);

create table if not exists public.tuition_invoices (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.students(id) on delete cascade,
  invoice_number text not null unique,
  item text not null,
  amount_cents integer not null check (amount_cents >= 0),
  due_date date not null,
  status text not null default 'due' check (status in ('due','overdue','paid','void')),
  paid_at timestamptz,
  created_at timestamptz not null default now()
);

create table if not exists public.academy_updates (
  id uuid primary key default gen_random_uuid(),
  student_id uuid references public.students(id) on delete cascade,
  title text not null,
  body text not null,
  author_name text not null default 'BrightHive Academic Studio',
  category text not null default '课堂动态',
  image_url text,
  published_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- Teacher-maintained learning dashboard. Families can read linked students only.
create table if not exists public.student_dashboard_metrics (
  student_id uuid primary key references public.students(id) on delete cascade,
  average_score numeric(5,2) check (average_score between 0 and 100),
  homework_completion smallint check (homework_completion between 0 and 100),
  completed_homework integer check (completed_homework >= 0),
  subject_count integer check (subject_count >= 0),
  updated_at timestamptz not null default now()
);

create table if not exists public.student_score_trend (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.students(id) on delete cascade,
  label text not null,
  score numeric(5,2) not null check (score between 0 and 100),
  recorded_on date not null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.student_subject_performance (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.students(id) on delete cascade,
  subject text not null,
  score numeric(5,2) not null check (score between 0 and 100),
  updated_at timestamptz not null default now(),
  unique (student_id, subject)
);

create table if not exists public.student_learning_activities (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.students(id) on delete cascade,
  activity_type text not null,
  title text not null,
  teacher_name text not null,
  shared_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table if not exists public.student_homework_records (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.students(id) on delete cascade,
  has_homework boolean not null default true,
  completion_status text not null default 'unfinished' check (completion_status in ('unfinished','completed')),
  grade text,
  correct_count integer check (correct_count >= 0),
  wrong_count integer check (wrong_count >= 0),
  subject text not null,
  assigned_on date not null default current_date,
  teacher_name text not null,
  updated_at timestamptz not null default now()
);

-- Daily attendance is teacher-managed and family read-only.
create table if not exists public.student_attendance (
  id uuid primary key default gen_random_uuid(),
  student_id uuid not null references public.students(id) on delete cascade,

create or replace function public.handle_new_user()
returns trigger language plpgsql security definer set search_path = '' as $$
begin
  insert into public.profiles (id, full_name)
  values (new.id, coalesce(new.raw_user_meta_data ->> 'full_name', new.raw_user_meta_data ->> 'name', ''))
  on conflict (id) do nothing;
  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created after insert on auth.users
for each row execute function public.handle_new_user();

insert into public.profiles (id, full_name)
select id, coalesce(raw_user_meta_data ->> 'full_name', raw_user_meta_data ->> 'name', '')
from auth.users on conflict (id) do nothing;

create or replace function public.touch_updated_at()
returns trigger language plpgsql set search_path = '' as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop function if exists public.complete_parent_onboarding(text, text);
create function public.complete_parent_onboarding(p_student_name text, p_invite_code text)
returns void
language plpgsql
security definer
set search_path = ''
as $$
declare
  cleaned_name text := btrim(coalesce(p_student_name, ''));
  cleaned_code text := upper(btrim(coalesce(p_invite_code, '')));
begin
  if auth.uid() is null then
    raise exception 'Authentication required';
  end if;
  if char_length(cleaned_name) < 2 or char_length(cleaned_name) > 60 then
    raise exception 'Invalid name';
  end if;
  if cleaned_code <> 'GSLDI2' then
    raise exception 'Invalid invite code';
  end if;

  update public.profiles
  set onboarding_student_name = cleaned_name,
      onboarding_complete = true,
      onboarding_completed_at = now(),
      invite_code_used = cleaned_code
  where id = auth.uid();

  if not found then
    raise exception 'Profile not found';
  end if;
end;
$$;

revoke all on function public.complete_parent_onboarding(text, text) from public;
grant execute on function public.complete_parent_onboarding(text, text) to authenticated;

create or replace function public.handle_new_user()
returns trigger language plpgsql security definer set search_path = '' as $$
begin
  insert into public.profiles (id, full_name, email)
  values (new.id, coalesce(new.raw_user_meta_data ->> 'full_name', new.raw_user_meta_data ->> 'name', ''), new.email)

end;
$$;

create or replace function public.complete_account_onboarding(p_display_name text, p_invite_code text)
returns text
language plpgsql
security definer
set search_path = ''
as $$
declare
  cleaned_name text := btrim(coalesce(p_display_name, ''));
  cleaned_code text := upper(btrim(coalesce(p_invite_code, '')));
  account_role text;
  linked_student_id uuid;
begin
  if auth.uid() is null then raise exception 'Authentication required'; end if;
  if char_length(cleaned_name) < 2 or char_length(cleaned_name) > 60 then raise exception 'Invalid name'; end if;

  if encode(extensions.digest(cleaned_code, 'sha256'), 'hex') = '4bc2da4c88a5d7f19e1a47aa428a1673cf6c387366839b10e4d87f9cf35ff6d2' then
    account_role := 'teacher';
    insert into public.staff_users (user_id, role, display_name)
    values (auth.uid(), 'teacher', cleaned_name)
    on conflict (user_id) do update set display_name = excluded.display_name;
  elsif cleaned_code = 'GSLDI2' then
    account_role := 'parent';
  else
    raise exception 'Invalid invite code';
  end if;

  update public.profiles
  set full_name = cleaned_name,
      role = account_role,
      onboarding_student_name = case when account_role = 'parent' then cleaned_name else null end,
      onboarding_complete = true,
      onboarding_completed_at = now(),
      invite_code_used = null
  where id = auth.uid();
  if not found then raise exception 'Profile not found'; end if;

  if account_role = 'parent' then
    select ps.student_id into linked_student_id
    from public.parent_students ps
    where ps.parent_id = auth.uid()
    limit 1;
    if linked_student_id is null then
      insert into public.students (first_name, last_name, grade, status)
      values (cleaned_name, '', '暂无', 'active')
      returning id into linked_student_id;
      insert into public.parent_students (parent_id, student_id, relationship)
      values (auth.uid(), linked_student_id, 'Parent');
    end if;
  end if;
  return account_role;
end;
$$;

-- Make every existing registered parent account immediately available in the teacher dashboard.
do $$
declare
  account_record record;
  linked_student_id uuid;
begin
  for account_record in
    select p.id, coalesce(nullif(btrim(p.onboarding_student_name), ''), nullif(btrim(p.full_name), ''), 'Student') as student_name
    from public.profiles p
    where p.role = 'parent'
      and not exists (select 1 from public.parent_students ps where ps.parent_id = p.id)
  loop
    insert into public.students (first_name, last_name, grade, status)
    values (account_record.student_name, '', '暂无', 'active')
    returning id into linked_student_id;
    insert into public.parent_students (parent_id, student_id, relationship)
    values (account_record.id, linked_student_id, 'Parent');
  end loop;
end;
$$;
